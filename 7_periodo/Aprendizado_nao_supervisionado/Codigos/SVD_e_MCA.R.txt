### SVD para Compressão de Imagem

library(jpeg)
library(ggplot2)

compress_image <- function(image_path, k) {
  # Carregar a imagem e converter para tons de cinza
  image <- readJPEG(image_path)
  image_gray <- 0.2989 * image[,,1] + 0.5870 * image[,,2] + 0.1140 * image[,,3]
  
  # Aplicar SVD
  svd_res <- svd(image_gray)
  
  # Reconstruir a imagem com k componentes
  compressed_image <- svd_res$u[, 1:k] %*% diag(svd_res$d[1:k]) %*% t(svd_res$v[, 1:k])
  
  # Normalizar a imagem reconstruída para ficar no intervalo [0,1]
  compressed_image <- pmax(pmin(compressed_image, 1), 0)
  
  list(original = image_gray, compressed = compressed_image)
}

dados_estudantes <- data.frame(
  estudante  = LETTERS[1:20],

  Trabalha   = c("Sim","Sim","Sim","Sim","Sim","Sim","Sim","Sim","Nao","Nao",

)

dados_estudantes


# ------------------------------------------------------------
# 2. Tabela de contingência: Período × Trabalha
# ------------------------------------------------------------
sjt.xtab(var.row = dados_estudantes$Periodo,
         var.col = dados_estudantes$Trabalha,
         show.exp  = TRUE,
         encoding  = "UTF-8")

tab_periodo_trabalha <- table(dados_estudantes$Periodo,
                              dados_estudantes$Trabalha)


# ------------------------------------------------------------
# 3. Teste qui-quadrado: Período × Trabalha
# ------------------------------------------------------------
# H₀: Período e hábito de trabalho são independentes
# H₁: Há associação entre Período e hábito de trabalho

qui2 <- chisq.test(x = tab_periodo_trabalha, correct = FALSE)
qui2

# V de Cramér — intensidade da associação
CramerV(tab_periodo_trabalha)


# ------------------------------------------------------------
# 4. Todos os pares: qui-quadrado + V de Cramér
# ------------------------------------------------------------
variaveis  <- colnames(dados_estudantes)[-1]  # exclui 'estudante'
resultados <- list()

for (i in 1:(length(variaveis) - 1)) {
  for (j in (i + 1):length(variaveis)) {
    tab <- table(dados_estudantes[[ variaveis[i] ]],
                 dados_estudantes[[ variaveis[j] ]])
    resultados[[ paste(variaveis[i], "vs", variaveis[j]) ]] <- list(
      V_Cramer      = round(CramerV(tab), 3),
      Qui_Quadrado  = round(chisq.test(tab, correct = FALSE)$statistic, 3),
      p_valor       = round(chisq.test(tab, correct = FALSE)$p.value,   4)
    )
  }
}

# Exibir como tabela
do.call(rbind, lapply(resultados, as.data.frame)) |>
  tibble::rownames_to_column("Par") |>
  kable(digits = 3, caption = "Associações entre pares de variáveis")
