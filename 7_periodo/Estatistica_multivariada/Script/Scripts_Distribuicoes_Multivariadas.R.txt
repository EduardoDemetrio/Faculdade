### A função a seguir constrói o gráfico qqplot com envelopes simulados
qqplot_bandas <- function(dados, n_simul = 1000, xbar = NULL, S = NULL,
                          probs = c(0.025, 0.50, 0.975),
                          pch = 20, lwd = 1.2,
                          col_mediana = "red", lty_mediana = 2) {

    stopifnot(is.data.frame(dados) || is.matrix(dados))
    dados <- as.matrix(dados)
    
    if (!is.numeric(dados)) stop("`dados` deve conter apenas colunas numéricas.")
    if (any(!is.finite(dados))) stop("`dados` contém NA/Inf. Remova/impute antes de usar.")
    
    n <- nrow(dados)
    p <- ncol(dados)
    
    if (is.null(xbar)) xbar <- colMeans(dados)
    if (is.null(S))    S    <- stats::var(dados)
    
    # Inversa da covariância: 1 vez só
    Sinv <- solve(S)
    
    # Distância de Mahalanobis ao quadrado (vetorizada)
    d2_amostra <- mahalanobis(dados, center = xbar, cov = S)
    
    # Quantis teóricos qui-quadrado
    qchis <- qchisq((1:n - 1/2) / n, df = p)
    
    # Simulações sob normal multivariada com parâmetros estimados do próprio 'dados'
    # (você pode trocar por mu=xbar e Sigma=S, se quiser fixar exatamente)
    mu_sim <- xbar
    S_sim  <- S
    
    mat_simul <- matrix(NA_real_, nrow = n, ncol = n_simul)
    
    for (i in seq_len(n_simul)) {
        data_simul <- MASS::mvrnorm(n = n, mu = mu_sim, Sigma = S_sim)
        d2_sim <- mahalanobis(data_simul, center = mu_sim, cov = S_sim)
        mat_simul[, i] <- sort(d2_sim)
    }
    
    # Bandas (envelopes)
    q_low <- apply(mat_simul, 1, quantile, probs = probs[1], na.rm = TRUE)
    q_med <- apply(mat_simul, 1, quantile, probs = probs[2], na.rm = TRUE)
    q_hi  <- apply(mat_simul, 1, quantile, probs = probs[3], na.rm = TRUE)
    
    # Plot
    plot(range(qchis), range(c(q_low, q_hi, sort(d2_amostra))),
         type = "n", las = 1,
         xlab = expression(paste("Quantis te\u00f3ricos  ", chi[p]^2)),
         ylab = expression(paste("Dist\u00e2ncias de Mahalanobis  ", D^2)))
    
    lines(qchis, q_low, lwd = lwd)
    lines(qchis, q_hi,  lwd = lwd)
    lines(qchis, q_med, lwd = lwd, lty = lty_mediana, col = col_mediana)
    
    points(qchis, sort(d2_amostra), pch = pch)

}
