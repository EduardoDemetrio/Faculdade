require(ggplot2)
require(GGally)
require(ggpubr)
require(klaR)
require(rattle)

options(device = 'x11')
setwd(this.path::here())

load('Dados_AD_Bancos.Rdata')
dados <- Dados_AD_Bancos

summary(dados)

### Nesta aplicação, o objetivo é discriminar (classificar) bancos "sem problemas"
### e bancos "com problemas" (sob investigação do Banco Central). As variáveis
### usadas na análise são as seguintes:

# Y1: Liquidez imediata;
# Y2: Participação dos depósitos interfinanceiros no total operacional;
# Y3: Participação das exigibilidades no ativo operacional;
# Y4: Participação das rendas de prestação de serviços em relação às despesas administrativas.

dados$Condição <- relevel(dados$Condição, ref = 'Sem problemas')

g1 <- ggplot(data = dados, aes(x = Condição, y = Y1, fill = Condição))+
  geom_boxplot()+
  theme_bw(base_size = 14)

g2 <- ggplot(data = dados, aes(x = Condição, y = Y2, fill = Condição))+
  geom_boxplot()+
  theme_bw(base_size = 14)

g3 <- ggplot(data = dados, aes(x = Condição, y = Y3, fill = Condição))+
  geom_boxplot()+
  theme_bw(base_size = 14)

g4 <- ggplot(data = dados, aes(x = Condição, y = Y4, fill = Condição))+
  geom_boxplot()+
  theme_bw(base_size = 13)

ggarrange(g1, g2, g3, g4, nrow = 2, ncol = 2, common.legend = TRUE, legend = 'none')
### No geral, as variáveis aparentemente são úteis para discriminar bancos nas
### duas condições.

ggpairs(dados, columns = 3:6, ggplot2::aes(colour=Condição)) + theme_bw(base_size = 13)
### Matriz de gráficos de dispersão.

ggplot(data = dados, aes(x = Y1, y = Y2, col = Condição))+
    geom_point(size = 3)+
    theme_bw(base_size = 14)+
    theme(legend.position = 'bottom')
### Observe como evidentemente os dois grupos são linearmente discriminados por
### Y1 e Y2.

### Análise discriminante linear
ajuste <- lda(Condição ~ Y1 + Y2 + Y3 + Y4, data = dados)
ajuste
### A função discriminante é dada por LD = 1.914*Y1 + 3.950 * Y2 + 0.197 * Y3 + 4.334 * Y4.

pred_linear <- predict(ajuste)
pred_linear$x
### Escores da função discriminante linear para os 20 bancos. Para o cálculo do
### escore, as quatro variáveis originais são centradas na média. Caso fossem
### adotadas probabilidades a priori diferentes para os dois desfechos, então
### as médias seriam ponderadas pelas prioris.

pred_linear$posterior
### Probabilidades a posteriori das observações pertencerem a cada grupo, baseadas
### nos valores das variáveis (Y1 a Y4) e nas probabilidades a priori.

pred_linear$class
### Classificação usando a função discriminante. Considerando os dados originais
### centrados, então a regra é classificar como "Com problemas" os bancos com LD>0
### e "Sem problemas" aqueles com LD<0 (ou, de maneira equivalente, com probabilidade
### a posteriori de ser "Com problemas" maior (menor) que 0.5.

plot(ajuste)
### Representação gráfica dos resultados da análise discriminante linear.

### Vamos construir a tabela de confusão
lda_class <- factor(pred_linear$class, labels = c('Pred_Sem', 'Pred_Com'))
table(dados$Condição, lda_class)
### Todos os 10 bancos sem problemas foram classificados como "sem problemas",
### enquanto dos 10 bancos com problemas, 9 foram classificados como "com problemas"
### e um "sem problema".

### Vamos usar a função discriminante linear para classificar quatro novos bancos.
novos_bancos <- data.frame(
  Banco = c('A', 'B', 'C', 'D'),
  Y1 = c(0.85, 0.52, 2.01, 0.49),
  Y2 = c(0.69, 0.95, 0.35, 0.32),
  Y3 = c(1.01, 1.12, 1.01, 1.11), 
  Y4 = c(0.37, 0.12, 0.07, 0.16)
)
novos_bancos

pred_novos <- predict(ajuste, newdata = novos_bancos[,2:5])
pred_novos
### Na saída, podemos ver as probabilidades a posteriori, o valor da função 
### discriminante linear, e as classificações para os quatro bancos. Podemos
### observar que os três primeiros bancos são classificados "sem problemas"
### e apenas o quarto "com problemas".

### Vamos incorporar agora probabilidades a priori, Vamos supor que tenhamos a
### informação, a priori, que 75% dos bancos não tem problemas, e 25% tem.

ajuste_prior <- lda(Condição ~ Y1 + Y2 + Y3 + Y4, prior = c(0.75,0.25),
                    data = dados)
ajuste_prior
### A função discriminante é a mesma do caso anterior.

pred_linear_prior <- predict(ajuste_prior)
pred_linear_prior$x
### Os escores da função linear discriminante são calculados como anteriormente,
### mas centrados na média ponderada pelas probabilidades a priori.

pred_linear_prior$posterior
### Probabilidades a posteriori das observações pertencerem a cada grupo, baseadas
### nos valores das variáveis (Y1 a Y4) e nas probabilidades a priori.

pred_linear_prior$class
### Classificação usando a função discriminante. Novamente, a regra é classificar 
### como "com problemas" os bancos com probabilidade a posteriori de ser "com problemas"
### maior que 0.5 e "sem problemas" caso contrário.

### Classificação dos novos bancos
pred_novos_prior <- predict(ajuste_prior, newdata = novos_bancos[,2:5])
pred_novos_prior
### Embora as probabilidades a posteriori sejam diferentes, as classificações
### coincidiram com as anteriores, obtidas sem o uso de prioris.

### Vamos usar validação cruzada (leave-one-out) para avaliar a classificação
### do modelo.
ajuste_CV <- lda(Condição ~ Y1 + Y2 + Y3 + Y4, data = dados, CV = TRUE)
ajuste_CV$posterior
### Probabilidades a posteriori para cada observação em cada classe.
ajuste_CV$class
### Classificações obtidas por validação cruzada.

### Vamos construir a tabela de confusão
lda_class <- factor(ajuste_CV$class, labels = c('Pred_Sem', 'Pred_Com'))
table(dados$Condição, lda_class)
### Observe que agora, usando validação cruzada, a acurácia estimada do modelo
### diminui, havendo dois bancos com problemas e dois bancos sem problemas
### classificados incorretamente.


### Vamos comparar com a performance preditiva da análise linear quadrática.
### Primeiramente, sem validação cruzada.
ajuste_quad <- qda(Condição ~ Y1 + Y2 + Y3 + Y4, data = dados)
ajuste_quad

pred_quad <- predict(ajuste_quad)
pred_quad$posterior
### Probabilidades a posteriori das observações pertencerem a cada grupo, baseadas
### nos valores das variáveis (Y1 a Y4) e nas probabilidades a priori.

pred_quad$class
### Classificação usando a função discriminante.

### Vamos construir a tabela de confusão
qda_class <- factor(pred_quad$class, labels = c('Pred_Sem', 'Pred_Com'))
table(dados$Condição, qda_class)
### Neste caso, temos dois bancos com problemas classificados incorretamente.


### Agora, a análise discriminante quadrática usando validação cruzada.
ajuste_quad_CV <- qda(Condição ~ Y1 + Y2 + Y3 + Y4, data = dados, CV = TRUE)
ajuste_quad_CV

### Vamos construir a tabela de confusão
qda_class_CV <- factor(ajuste_quad_CV$class, labels = c('Pred_Sem', 'Pred_Com'))
table(dados$Condição, qda_class_CV)
### Neste caso, temos cinco observações classificadas incorretamente (três sem
### problemas e duas com problemas).

### Agora, um problema de análise discriminante envolvendo três grupos (populações)

vinhos <- rattle::wine 
### Data-frame do pacote rattle.

head(vinhos, 10)
summary(vinhos)

ggpairs(vinhos, columns = 2:14, ggplot2::aes(colour=Type), 
        upper = list(continuous = wrap("cor", size = 2.5)))

### Vamos usar análise discriminante de Fisher para discriminar os tipos de
### vinho segundo as demais variáveis (características químicas).

ajuste_vinho <- lda(Type~., data = vinhos)
ajuste_vinho
### Neste caso, não estamos especificando probabilidades a priori, de forma que
### elas ficam definidas pelas próprias proporções amostrais. Além disso, não
### estamos usando validação cruzada.

### As duas funções discriminantes lineares são dadas por:
# LD1 = -0.403*Alcohol + 0.165*Malic ... - 0.002*Proline
# LD2 = 0.871*Alcohol + 0.305*Malic ... + 0.002*Proline

### Obtendo as predições
pred_vinho <- predict(ajuste_vinho)

pred_vinho$x
### Escores dos vinhos na amostra para as duas funções discriminantes lineares.

### Distribuição dos escores das funções discriminantes para os diferentes tipos de vinho.
ldahist(pred_vinho$x[,1], g = vinhos$Type)
ldahist(pred_vinho$x[,2], g = vinhos$Type)

plot(pred_vinho$x, col = vinhos$Type, cex = 1.2, pch = 20)
### Gráfico de dispersão para os escores das duas funções discriminantes lineares
### segundo o tipo de vinho.

pred_vinho$posterior ### Probabilidades a posteriori

pred_vinho$class ### Classificação dos vinhos baseada nas probabilidades a postewriori.

### Vamos construir a tabela de confusão
lda_class <- factor(pred_vinho$class, labels = c('Pred_A', 'Pred_B', 'Pred_C'))
table(vinhos$Type, lda_class)
### Neste caso, a análise linear discriminante não produziu qualquer erro de 
### classificação para os dados de ajuste.

### Vamos ver como ficaria a classificação usando validação cruzada

### Agora, a análise discriminante quadrática usando validação cruzada.
ajuste_vinho_CV <- lda(Type ~ ., data = vinhos, CV = TRUE)
ajuste_vinho_CV

### Vamos construir a tabela de confusão
vinho_class_CV <- factor(ajuste_vinho_CV$class, labels = c('Pred_A', 'Pred_B', 'Pred_C'))
table(vinhos$Type, vinho_class_CV)
