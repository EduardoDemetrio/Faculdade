################################################################################
### Análise Fatorial Exploratória
### Parte 3: métodos de extração, rotações e escores fatoriais
################################################################################

################################################################################
### Pacotes
################################################################################

require(psych)
require(corrplot)
require(ggplot2)
require(ggpubr)

################################################################################
### Configurações iniciais
################################################################################

options(device = "x11")

setwd(this.path::here())

################################################################################
### Base de dados
################################################################################
### A base bfi contém 25 itens de autorrelato de personalidade retirados do
### International Personality Item Pool.
###
### Os itens fazem parte do projeto SAPA, com informações de aproximadamente
### 2.800 indivíduos. Há também variáveis demográficas, como sexo,
### escolaridade e idade.
################################################################################

data(bfi)

help(bfi)

################################################################################
### Organização dos dados
################################################################################

dados_af <- na.omit(
    bfi[, c(1:25, 26, 28)]
)

################################################################################
### Separação das variáveis
################################################################################
### As 25 primeiras variáveis serão usadas na análise fatorial.
### Sexo e idade serão usadas posteriormente para interpretar os escores.
################################################################################

dados <- dados_af[, 1:25]

################################################################################
### Matriz de correlações
################################################################################

matcor <- round(
    cor(dados),
    3
)

corrplot(
    matcor,
    method = "color"
)

################################################################################
### Avaliação da adequação da análise fatorial
################################################################################

cortest.bartlett(
    matcor,
    n = nrow(dados)
)

################################################################################
### Interpretação
################################################################################
### O teste de Bartlett avalia a hipótese nula de que a matriz de correlações
### é igual à matriz identidade.
###
### Rejeitar essa hipótese indica que há correlações suficientes entre as
### variáveis para justificar uma análise fatorial.
################################################################################

KMO(matcor)$MSA

################################################################################
### Interpretação
################################################################################
### O índice KMO avalia a adequação global da matriz de correlações para
### análise fatorial.
###
### Valores mais próximos de 1 indicam maior adequação.
################################################################################

KMO(matcor)$MSAi

################################################################################
### Interpretação
################################################################################
### Os valores individuais de MSA indicam a adequação de cada variável à
### análise fatorial.
###
### Variáveis com valores baixos podem prejudicar a solução fatorial.
################################################################################


################################################################################
### Escolha do número de fatores
################################################################################

fa.parallel(
    matcor,
    n.obs = nrow(dados),
    fa = "pc"
)

################################################################################
### Interpretação
################################################################################
### A análise paralela sugere uma solução com cinco fatores.
###
### Essa escolha também é coerente com o contexto da base, construída para
### representar dimensões clássicas de personalidade.
################################################################################


################################################################################
### Solução inicial por componentes principais
################################################################################

f_pca <- principal(
    dados,
    nfactors = 5,
    rotate = "none"
)

################################################################################
### Principais resultados
################################################################################

f_pca$communality
f_pca$loadings
f_pca$fit
f_pca$uniquenesses
f_pca$residual
f_pca$rms

################################################################################
### Interpretação
################################################################################
### A solução por componentes principais é usada aqui.
###
################################################################################

corrplot.mixed(
    f_pca$residual,
    upper = "ellipse"
)

################################################################################
### Interpretação
################################################################################
### A matriz residual mostra as correlações que não foram bem reproduzidas
### pela solução com cinco fatores.
###
### Resíduos pequenos indicam que a estrutura fatorial reproduz bem a matriz
### de correlações observada.
################################################################################


################################################################################
### Comparação entre métodos de extração
################################################################################
### Nesta seção, mantemos fixo o número de fatores em cinco e não aplicamos
### rotação.
###
### O objetivo é comparar diferentes métodos de extração das cargas fatoriais.
################################################################################


################################################################################
### Método dos componentes principais
################################################################################

f_pca <- principal(
    dados,
    nfactors = 5,
    rotate = "none"
)

fa.diagram(f_pca)

f_pca$fit
f_pca$rms

################################################################################

    fm = "pa"

f_pa$fit
f_pa$rms


################################################################################
### Método MINRES
################################################################################
### O método MINRES busca minimizar os resíduos entre a matriz de correlações
### observada e a matriz de correlações reproduzida pelo modelo fatorial.
################################################################################

f_mr <- fa(
    dados,
    nfactors = 5,
    rotate = "none",
    fm = "minres"
)

fa.diagram(f_mr)

f_mr$fit
f_mr$rms


################################################################################
### Método de mínimos quadrados ponderados
################################################################################

fa.diagram(f_wls)

f_wls$fit
f_wls$rms

################################################################################
### Interpretação
################################################################################
### Diferentes métodos podem produzir soluções semelhantes, mas não
### necessariamente idênticas.
################################################################################


################################################################################

################################################################################

################################################################################
### Nas rotações oblíquas, os fatores podem ser correlacionados.
###
### Isso pode ser mais realista em muitas aplicações, especialmente quando
### os construtos latentes não são teoricamente independentes.
###
### A matriz Phi apresenta as correlações entre os fatores.
################################################################################

data_escores <- data.frame(
    f_reg$scores,
    Sexo = dados_af$gender,
    Idade = dados_af$age
)

data_escores$Sexo <- factor(
    data_escores$Sexo,
    labels = c("Masculino", "Feminino")
)

################################################################################
### Fator 1
################################################################################

g1 <- ggplot(
    data = data_escores,
    aes(
        x = Sexo,
        y = MR1,
        fill = Sexo
    )
) +
    geom_boxplot() +
    theme_bw(base_size = 14) +
    theme(
        legend.position = "none"
    ) +
    labs(
        x = NULL,
        y = "Fator 1"
    )

g2 <- ggplot(
    data = data_escores,
    aes(
        x = Idade,
        y = MR1
    )
) +
    geom_point(alpha = 0.35) +
    geom_smooth(
        method = "lm",
        se = TRUE
    ) +
    theme_bw(base_size = 14) +
    labs(
        x = "Idade",
        y = "Fator 1"
    )

ggarrange(
    g1,
    g2,
    nrow = 1,
    ncol = 2
)
