data("mtcars")

dados_bivariados <- mtcars[, c("wt", "disp")]

# Gráfico de dispersão original
ggplot(dados_bivariados, aes(x = wt, y = disp)) +
    geom_point(size = 2) +
    theme_bw(base_size = 14) +
    labs(x = "wt", y = "disp")

# Matrizes básicas
cov(dados_bivariados)   # Matriz de covariâncias
cor(dados_bivariados)   # Matriz de correlações

# ACP baseada na matriz de covariâncias
acp_cov_biv <- princomp(dados_bivariados, cor = FALSE)
summary(acp_cov_biv)
loadings(acp_cov_biv)

# ACP baseada na matriz de correlações
acp_cor_biv <- princomp(dados_bivariados, cor = TRUE)
summary(acp_cor_biv)
loadings(acp_cor_biv)

# Representação gráfica no espaço padronizado
dados_biv_pad <- data.frame(
    wt = as.numeric(scale(dados_bivariados$wt)),
    disp = as.numeric(scale(dados_bivariados$disp))
)

ggplot(dados_biv_pad, aes(x = wt, y = disp)) +
    geom_point(size = 2) +
    geom_abline(intercept = 0, slope = 0.707, color = "red",
                linetype = 2, linewidth = 1) +
    geom_abline(intercept = 0, slope = -0.707, color = "red",
                linetype = 2, linewidth = 1) +
    theme_bw(base_size = 14) +
    labs(x = "wt padronizado", y = "disp padronizado")+
    coord_equal()

data("dec", package = "combinedevents")

dados_decatlo <- dec[, c("100m_p", "LJ_p", "SP_p", "HJ_p",
                         "400m_p", "110mH_p", "DT_p", "PV_p",
                         "JT_p", "1500m_p")]

names(dados_decatlo) <- str_remove(names(dados_decatlo), "_p")

rownames(dados_decatlo) <- c(
    "A. Eaton", "K. Mayer", "D. Warner",
    "K. Kazmirek", "L. Bourrada", "L. Suarez",
    "Z. Ziemek", "T. Plaetsen", "K. Felix",
    "L. Araujo", "J. Taiwo", "A. Helcelet",
    "B. Auzeil", "C. Dubler", "A. Abele",
    "L. Victor", "P. Tonnesen", "Y. Garcia",
    "D. Distelberger", "K. Ushiro", "P. Wiesiolek",
    "A. Nakamura", "K. Saluri"
)

# Exploração descritiva
ggpairs(dados_decatlo) +
    theme_bw(base_size = 14)

ggcorrplot(cor(dados_decatlo),
           colors = c("#E46726", "white", "#6D9EC1"),
           hc.order = TRUE,
           lab = TRUE)

mat_cov_dec <- var(dados_decatlo)
mat_cor_dec <- cor(dados_decatlo)

# Número de componentes - matriz de covariâncias
eigen_cov_dec <- eigen(mat_cov_dec)
eigen_cov_dec
eigen_cov_dec$values
eigen_cov_dec$vectors
mean(eigen_cov_dec$values)

plot(1:10, eigen_cov_dec$values, type = "b", pch = 20,
     xlab = "Componente", ylab = "Autovalor")
axis(1, at = 1:10)
abline(h = mean(eigen_cov_dec$values), lty = 2, col = "red")

# Número de componentes - matriz de correlações
eigen_cor_dec <- eigen(mat_cor_dec)
eigen_cor_dec
eigen_cor_dec$values
eigen_cor_dec$vectors

plot(1:10, eigen_cor_dec$values, type = "b", pch = 20,
     xlab = "Componente", ylab = "Autovalor")
axis(1, at = 1:10)
abline(h = 1, lty = 2, col = "red")

# Análise paralela
fa.parallel(mat_cor_dec, n.obs = nrow(dados_decatlo), fa = "pc")

# ACP baseada na matriz de covariâncias
acp_cov_dec <- princomp(dados_decatlo, cor = FALSE, scores = TRUE)

print(acp_cov_dec)
summary(acp_cov_dec)
print(acp_cov_dec$loadings, cutoff = 0)

# Correlações variáveis x CP1 e CP2 (ACP por covariâncias)
cor_cp1_cov_dec <- loadings(acp_cov_dec)[, 1] *
    sqrt(eigen_cov_dec$values[1]) / sqrt(diag(mat_cov_dec))

cor_cp2_cov_dec <- loadings(acp_cov_dec)[, 2] *
    sqrt(eigen_cov_dec$values[2]) / sqrt(diag(mat_cov_dec))

cor_cp1_cov_dec
cor_cp2_cov_dec

acp_cov_dec$scores ### Escores calculados para cada atleta com base nos CPs obtidos

ggpairs(data.frame(acp_cov_dec$scores[, 1:3])) +
    theme_bw(base_size = 14)

# ACP baseada na matriz de correlações
acp_cor_dec <- princomp(dados_decatlo, cor = TRUE, scores = TRUE)

print(acp_cor_dec)
summary(acp_cor_dec)
print(acp_cor_dec$loadings, cutoff = 0)

# Correlações variáveis x CP1 e CP2 (ACP por correlações)
cor_cp1_cor_dec <- loadings(acp_cor_dec)[, 1] * sqrt(eigen_cor_dec$values[1])
cor_cp2_cor_dec <- loadings(acp_cor_dec)[, 2] * sqrt(eigen_cor_dec$values[2])

cor_cp1_cor_dec
cor_cp2_cor_dec

acp_cor_dec$scores ### Escores calculados para cada atleta com base nos CPs obtidos

# Biplot CP1 x CP2
fviz_pca_biplot(acp_cor_dec,
                col.var = "darkblue",
                col.ind = "blue")

# Biplot CP1 x CP3
fviz_pca_biplot(acp_cor_dec,
                col.var = "darkblue",
                col.ind = "blue",
                axes = c(1, 3))
