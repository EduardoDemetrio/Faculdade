################################################################################
### Teste de permutação para T2 de Hotelling - caso de uma amostra
### H0: mu = mu0
### Método: permutação por trocas de sinal
################################################################################

################################################################################
### Vamos retomar os dados de composição de solo

setwd(this.path::here())

data_T2_uni <- read.csv2('UD04_Base_T2_Uma_Populacao.csv', 
                         fileEncoding = 'CP1252', 
                         stringsAsFactors = TRUE, dec = '.')

dados <- data_T2_uni[,c('Areia', 'Argila')]

mu0 <- c(14, 42)

################################################################################
### Matriz de dados

Y <- as.matrix(dados)
n <- nrow(Y)
p <- ncol(Y)

################################################################################
### Centralização em torno de mu0

X <- sweep(Y, 2, mu0, FUN = "-")

################################################################################
### Estatística T2 observada

x_bar <- colMeans(X)
S <- cov(X)

T2_obs <- n * t(x_bar) %*% solve(S) %*% x_bar
T2_obs <- as.numeric(T2_obs)

T2_obs

################################################################################
### Permutação por sinais aleatórios
### Sob H0 e simetria em torno de 0, X_i e -X_i são equiprováveis

set.seed(2026)

B <- 9999
T2_perm <- numeric(B)

for (b in 1:B) {
    
    sinais <- sample(c(-1, 1), size = n, replace = TRUE)
    
    X_b <- X * sinais
    
    x_bar_b <- colMeans(X_b)
    S_b <- cov(X_b)
    
    T2_perm[b] <- n * t(x_bar_b) %*% solve(S_b) %*% x_bar_b
}

################################################################################
### Histograma da distribuição empírica

hist(T2_perm,
     breaks = 40,
     main = expression("Distribuição permutacional de " * T^2),
     xlab = expression(T^2))

abline(v = T2_obs, lwd = 2, lty = 2)

################################################################################
### p-valor empírico

p_valor <- (sum(T2_perm >= T2_obs) + 1) / (B + 1)

p_valor


################################################################################
### Teste de permutação para T2 de Hotelling - caso de duas amostras
### H0: muA = muN
### Método: permutação por trocas de sinal


data_T2_npar_igual <- read.csv2('UD04_Base_T2_Duas_Populacoes_Independente_Variancias_Iguais.csv', 
                                fileEncoding = 'CP1252', 
                                stringsAsFactors = TRUE, dec = '.')

################################################################################
### Estatística T2 observada

grupo <- ifelse(data_T2_npar_igual$Curso == 'Antigo', 1, 2)

teste <- TwoSamplesHT2(data_T2_npar_igual[,c('Gramática', 'Audição', 'Conversação')], 
              group = grupo, 
              alpha = 0.05, Homogenity = TRUE)

T2_obs <- teste$HT2

################################################################################
### Permutação por aleatorização dos labels dos grupos (1 e 2)
### Sob H0 as atribuições aleatórias dos labels são equiprováveis

B <- 9999
T2_perm <- numeric(B)

for(b in 1:B){
    grupo_permut <- sample(grupo)
    teste <- TwoSamplesHT2(data_T2_npar_igual[,c('Gramática', 'Audição', 'Conversação')], 
                           group = grupo_permut, 
                           alpha = 0.05, Homogenity = TRUE)
    T2_perm[b] <- teste$HT2
}


################################################################################
### Histograma da distribuição empírica

hist(T2_perm,
     breaks = 40,
     main = expression("Distribuição permutacional de " * T^2),
     xlab = expression(T^2))

abline(v = T2_obs, lwd = 2, lty = 2)


################################################################################
### p-valor empírico

p_valor <- (sum(T2_perm >= as.numeric(T2_obs)) + 1) / (B + 1)
p_valor
